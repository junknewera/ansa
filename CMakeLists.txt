cmake_minimum_required(VERSION 3.25)
project(ansa VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(YYJSON REQUIRED IMPORTED_TARGET yyjson)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(omh_core src/json.cpp src/sse.cpp src/responses.cpp src/tools.cpp src/transport.cpp
  src/runtime.cpp src/offline.cpp src/journal.cpp src/session.cpp src/compaction.cpp
  src/auth.cpp src/auth_crypto.cpp src/context_codec.cpp src/local_storage.cpp src/process.cpp src/agent_config.cpp src/local_tools.cpp)
target_include_directories(omh_core PUBLIC include)
target_link_libraries(omh_core PUBLIC PkgConfig::CURL PkgConfig::YYJSON OpenSSL::Crypto Threads::Threads)
target_compile_options(omh_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
option(OMH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(OMH_SANITIZERS)
  target_compile_options(omh_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(omh_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(ansa src/main.cpp src/cli.cpp src/auth_cli.cpp src/agent_cli.cpp)
target_link_libraries(ansa PRIVATE omh_core)
include(CTest)
option(OMH_MUTATION_CHILD "Disable recursive mutation test registration in isolated test builds" OFF)
if(BUILD_TESTING)
  # M0 environment probe is separate from the existing C++ parity baseline.
  add_executable(ansa_preflight tests/parity/preflight.cpp)
  add_executable(ansa_risk_oracle tests/parity/cpp_adapter.cpp)
  target_link_libraries(ansa_risk_oracle PRIVATE omh_core)
  add_executable(omh_tests tests/protocol_tests.cpp)
  target_link_libraries(omh_tests PRIVATE omh_core)
  target_compile_definitions(omh_tests PRIVATE OMH_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  foreach(suite dependencies sse reducer replay tools transport contracts)
    add_test(NAME ${suite} COMMAND omh_tests ${suite})
  endforeach()
  set_tests_properties(dependencies sse reducer replay tools transport contracts PROPERTIES TIMEOUT 60)
  add_test(NAME cli-self-check COMMAND ansa --self-check)
  add_test(NAME offline-roundtrip COMMAND ansa --offline-demo)
  add_test(NAME cli-offline-jsonl COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/offline_jsonl.sh" $<TARGET_FILE:ansa>)
  add_test(NAME cli-session-jsonl COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/session_jsonl.sh" $<TARGET_FILE:ansa>)
  set_tests_properties(offline-roundtrip cli-offline-jsonl cli-session-jsonl PROPERTIES TIMEOUT 30)
  add_executable(omh_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(omh_runtime_tests PRIVATE omh_core)
  target_compile_definitions(omh_runtime_tests PRIVATE OMH_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  foreach(suite cancel priority admission resync invalid-response tool-cancel roundtrip shutdown)
    add_test(NAME runtime-${suite} COMMAND omh_runtime_tests ${suite})
    set_tests_properties(runtime-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(omh_session_tests tests/session_tests.cpp)
  target_link_libraries(omh_session_tests PRIVATE omh_core)
  foreach(suite journal conflict history crash storage-failure blocked-disk late-outcome compaction compact-failure
      resume-repeat item-budget intent-cancel compact-cancel compact-budget parts-limit batch-recovery dispatch-guard)
    add_test(NAME session-${suite} COMMAND omh_session_tests ${suite})
    set_tests_properties(session-${suite} PROPERTIES TIMEOUT 60)
  endforeach()
  set_tests_properties(session-item-budget PROPERTIES TIMEOUT 180)
  add_executable(omh_auth_tests tests/auth_tests.cpp)
  target_link_libraries(omh_auth_tests PRIVATE omh_core)
  foreach(suite jwt lifecycle permission refresh-crash pending-cleanup dispatch-logout
      logout-login logout-refresh logout-crash logout-marker process-refresh refresh-scope uncertain-login held-dispatch
      committed-login-crash execution-logout)
    add_test(NAME auth-${suite} COMMAND omh_auth_tests ${suite})
    set_tests_properties(auth-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(omh_summary_tests tests/summary_tests.cpp)
  target_link_libraries(omh_summary_tests PRIVATE omh_core)
  foreach(suite codec runtime cancel)
    add_test(NAME summary-${suite} COMMAND omh_summary_tests ${suite})
    set_tests_properties(summary-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  find_program(ANSA_TEST_SHELL NAMES sh REQUIRED)
  add_executable(ansa_loop_tests tests/agent_loop_tests.cpp)
  target_link_libraries(ansa_loop_tests PRIVATE omh_core)
  target_compile_definitions(ansa_loop_tests PRIVATE OMH_TEST_SHELL="${ANSA_TEST_SHELL}")
  foreach(suite roundtrip known-failure limit crash between-rounds process-recovery)
    add_test(NAME agent-loop-${suite} COMMAND ansa_loop_tests ${suite})
    set_tests_properties(agent-loop-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(ansa_process_tests tests/process_tests.cpp)
  target_link_libraries(ansa_process_tests PRIVATE omh_core)
  target_compile_definitions(ansa_process_tests PRIVATE OMH_TEST_SHELL="${ANSA_TEST_SHELL}")
  foreach(suite basic output cancel timeout privileges crash detached evidence signals guardian-crash)
    add_test(NAME process-${suite} COMMAND ansa_process_tests ${suite})
    set_tests_properties(process-${suite} PROPERTIES TIMEOUT 15)
  endforeach()
  add_executable(ansa_local_tests tests/local_tests.cpp)
  target_link_libraries(ansa_local_tests PRIVATE omh_core)
  foreach(suite config files catalog instruction-links)
    add_test(NAME local-${suite} COMMAND ansa_local_tests ${suite})
    set_tests_properties(local-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(ansa_cli_tests tests/agent_cli_tests.cpp)
  target_link_libraries(ansa_cli_tests PRIVATE omh_core)
  foreach(suite backpressure pending-cancel busy input-size terminal)
    add_test(NAME agent-cli-${suite} COMMAND ansa_cli_tests ${suite})
    set_tests_properties(agent-cli-${suite} PROPERTIES TIMEOUT 15)
  endforeach()
  add_executable(ansa_frontend_tests tests/frontend_runtime_tests.cpp)
  target_link_libraries(ansa_frontend_tests PRIVATE omh_core)
  foreach(suite streaming recovery adversarial modes overflow)
    add_test(NAME frontend-${suite} COMMAND ansa_frontend_tests ${suite})
    set_tests_properties(frontend-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  if(NOT OMH_MUTATION_CHILD)
    foreach(mode drop reorder)
      add_test(NAME mutation-${mode} COMMAND ${CMAKE_COMMAND}
        -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR}
        -DMODE=${mode} -DGENERATOR=${CMAKE_GENERATOR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/mutation.cmake)
      set_tests_properties(mutation-${mode} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    endforeach()
  endif()
endif()
install(TARGETS ansa RUNTIME DESTINATION bin)
