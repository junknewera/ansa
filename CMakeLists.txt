cmake_minimum_required(VERSION 3.25)
project(oh-my-harness VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(YYJSON REQUIRED IMPORTED_TARGET yyjson)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(omh_core src/json.cpp src/sse.cpp src/responses.cpp src/tools.cpp src/transport.cpp
  src/runtime.cpp src/offline.cpp src/journal.cpp src/session.cpp src/compaction.cpp)
target_include_directories(omh_core PUBLIC include)
target_link_libraries(omh_core PUBLIC PkgConfig::CURL PkgConfig::YYJSON OpenSSL::Crypto Threads::Threads)
target_compile_options(omh_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
option(OMH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(OMH_SANITIZERS)
  target_compile_options(omh_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(omh_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(omh src/main.cpp src/cli.cpp)
target_link_libraries(omh PRIVATE omh_core)
include(CTest)
option(OMH_MUTATION_CHILD "Disable recursive mutation test registration in isolated test builds" OFF)
if(BUILD_TESTING)
  add_executable(omh_tests tests/protocol_tests.cpp)
  target_link_libraries(omh_tests PRIVATE omh_core)
  target_compile_definitions(omh_tests PRIVATE OMH_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  foreach(suite dependencies sse reducer replay tools transport contracts)
    add_test(NAME ${suite} COMMAND omh_tests ${suite})
  endforeach()
  set_tests_properties(dependencies sse reducer replay tools transport contracts PROPERTIES TIMEOUT 60)
  add_test(NAME cli-self-check COMMAND omh --self-check)
  add_test(NAME offline-roundtrip COMMAND omh --offline-demo)
  add_test(NAME cli-offline-jsonl COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/offline_jsonl.sh" $<TARGET_FILE:omh>)
  add_test(NAME cli-session-jsonl COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/session_jsonl.sh" $<TARGET_FILE:omh>)
  set_tests_properties(offline-roundtrip cli-offline-jsonl cli-session-jsonl PROPERTIES TIMEOUT 30)
  add_executable(omh_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(omh_runtime_tests PRIVATE omh_core)
  target_compile_definitions(omh_runtime_tests PRIVATE OMH_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  foreach(suite cancel priority admission resync invalid-response tool-cancel roundtrip shutdown)
    add_test(NAME runtime-${suite} COMMAND omh_runtime_tests ${suite})
    set_tests_properties(runtime-${suite} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(omh_session_tests tests/session_tests.cpp)
  target_link_libraries(omh_session_tests PRIVATE omh_core)
  foreach(suite journal conflict history crash storage-failure blocked-disk late-outcome compaction compact-failure
      resume-repeat item-budget intent-cancel compact-cancel compact-budget parts-limit batch-recovery)
    add_test(NAME session-${suite} COMMAND omh_session_tests ${suite})
    set_tests_properties(session-${suite} PROPERTIES TIMEOUT 60)
  endforeach()
  set_tests_properties(session-item-budget PROPERTIES TIMEOUT 180)
  if(NOT OMH_MUTATION_CHILD)
    foreach(mode drop reorder)
      add_test(NAME mutation-${mode} COMMAND ${CMAKE_COMMAND}
        -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR}
        -DMODE=${mode} -DGENERATOR=${CMAKE_GENERATOR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/mutation.cmake)
      set_tests_properties(mutation-${mode} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    endforeach()
  endif()
endif()
install(TARGETS omh RUNTIME DESTINATION bin)
